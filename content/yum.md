    - installed                       # 列已安装
    - extras                          # 已安装但不在yum repository的包

- install
- remove
- info                                # 包详情
    - updates
    - installed
    - extras
- provides                            # 包文件
- whatprovides 'bin/isstat'           # 查看命令属性哪个包
- history
    - list
    - redo 序号                        # 重新做序号
    - undo 序号                        # 恢复历史中执行的动作
- groups list                         # 查看安装的组

    - 或
    - proxy=FTP://XXX.XXX.XXX.XXX:XXXX
    - proxy_username=你的用户名
    - proxy_password=你的用户名的密码
